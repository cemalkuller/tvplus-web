$CREATE_RELEASE()

cd "$FORGE_RELEASE_DIRECTORY"

npm install --omit=dev --no-audit --no-fund

(
    set -eu
    DUMP_FILE="$FORGE_RELEASE_DIRECTORY/tvplus_db.sql"
    DONE_FILE="$FORGE_SITE_ROOT/.tvplus_db.imported"

    if [ ! -f "$DONE_FILE" ]; then
        test -f "$DUMP_FILE" || { echo "SQL dump missing: $DUMP_FILE"; exit 1; }
        : "${DB_USER:?Set DB_USER in Forge Environment}"
        : "${DB_PASSWORD:?Set DB_PASSWORD in Forge Environment}"
        export MYSQL_PWD="$DB_PASSWORD"
        MYSQL_ARGS=(--host="${DB_HOST:-127.0.0.1}" --port="${DB_PORT:-3306}" --user="$DB_USER")
        TABLE_COUNT=$(mysql "${MYSQL_ARGS[@]}" --batch --skip-column-names -e "SELECT COUNT(*) FROM information_schema.tables WHERE table_schema='tvplus_db';")
        if [ "$TABLE_COUNT" != "0" ]; then
            echo "Import stopped: tvplus is not empty. Existing data was preserved."
            exit 1
        fi
        mysql "${MYSQL_ARGS[@]}" tvplus < "$DUMP_FILE"
        touch "$DONE_FILE"
        chmod 600 "$DUMP_FILE" "$DONE_FILE"
        echo "Database import completed. Future deployments will skip it."
    fi
)
